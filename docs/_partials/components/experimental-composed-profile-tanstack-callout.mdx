<If sdk="tanstack-react-start">
  > [!NOTE]
  > TanStack Start's Vite optimizer can pre-bundle `@clerk/ui/experimental` separately and inline a second copy of Clerk's React context, which surfaces as `useClerk can only be used within the <ClerkProvider /> component` on pages that render these panels. To force a single instance, add `@clerk/shared` as a direct dependency and update your `vite.config.ts`:
  >
  > ```ts {{ filename: 'vite.config.ts' }}
  > export default defineConfig({
  >   resolve: {
  >     dedupe: ['react', 'react-dom', '@clerk/shared', '@clerk/react'],
  >   },
  >   optimizeDeps: {
  >     include: ['@clerk/ui/experimental', '@clerk/shared/react'],
  >   },
  >   // ...plugins
  > })
  > ```
</If>
