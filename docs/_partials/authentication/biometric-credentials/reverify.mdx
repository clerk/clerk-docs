import { isBiometricCredentialError, useBiometricCredentials } from '@clerk/expo'
import { Button } from 'react-native'

export function ReverifyWithBiometrics() {
  const { reverify } = useBiometricCredentials()

  const handleReverify = async () => {
    try {
      const result = await reverify({
        level: 'multi_factor',
        reason: 'Verify your identity to continue.',
      })

      if (result.status !== 'complete' || result.level !== 'multi_factor') {
        console.error(
          'Multi-factor requirement not met. Check MFA enrollment and verification options.',
        )
        return
      }

      console.log('Multi-factor verification complete')
    } catch (err) {
      if (isBiometricCredentialError(err)) {
        if (err.code === 'biometric_authentication_canceled') return

        if (err.code === 'biometric_credential_policy_incompatible') {
          console.error('Choose another verification method for this credential.')
          return
        }
      }

      console.error(JSON.stringify(err, null, 2))
    }
  }

  return <Button title="Verify with biometrics" onPress={handleReverify} />
}
