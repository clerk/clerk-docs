### Authorization checks (roles/permissions/features/plans) {{ toc: false }}

`<Protect>`'s `role`, `permission`, `feature`, and `plan` props become a single object passed to `<Show>`'s `when` prop. The `fallback` prop — or fallback slot in Astro, Vue, and Nuxt — carries over unchanged.

<If sdk={["nextjs", "react", "react-router", "tanstack-react-start", "chrome-extension", "expo", "astro"]}>
  ```tsx {{ del: [1, 4, 7, 10, 13, 16, 19, 22], ins: [2, 5, 8, 11, 14, 17, 20, 23], prettier: false }}
  <Protect role="admin">
  <Show when={{ role: 'admin' }}>
      <AdminPanel />
  </Protect>
  </Show>

  <Protect permission="org:billing:manage">
  <Show when={{ permission: 'org:billing:manage' }}>
      <BillingSettings />
  </Protect>
  </Show>

  <Protect feature="widgets">
  <Show when={{ feature: 'widgets' }}>
      <WidgetsPanel />
  </Protect>
  </Show>

  <Protect plan="pro">
  <Show when={{ plan: 'pro' }}>
      <ProFeatures />
  </Protect>
  </Show>
  ```
</If>

<If sdk={["vue", "nuxt"]}>
  ```vue {{ del: [2, 5, 8, 11, 14, 17, 20, 23], ins: [3, 6, 9, 12, 15, 18, 21, 24], prettier: false }}
  <template>
    <Protect role="admin">
    <Show :when="{ role: 'admin' }">
      <AdminPanel />
    </Protect>
    </Show>

    <Protect permission="org:billing:manage">
    <Show :when="{ permission: 'org:billing:manage' }">
      <BillingSettings />
    </Protect>
    </Show>

    <Protect feature="widgets">
    <Show :when="{ feature: 'widgets' }">
      <WidgetsPanel />
    </Protect>
    </Show>

    <Protect plan="pro">
    <Show :when="{ plan: 'pro' }">
      <ProFeatures />
    </Protect>
    </Show>
  </template>
  ```
</If>

### Bare `<Protect>` {{ toc: false }}

### `condition` callback {{ toc: false }}

If you were using `condition={(has) => ...}` on `<Protect>`, pass that callback to `when`:

<If sdk={["nextjs", "react", "react-router", "tanstack-react-start", "chrome-extension", "expo", "astro"]}>
  ```tsx {{ del: [1, 4], ins: [2, 5], prettier: false }}
  <Protect condition={(has) => has({ role: 'admin' }) && isAllowed}>
  <Show when={(has) => has({ role: 'admin' }) && isAllowed}>
      <AdminPanel />
  </Protect>
  </Show>
  ```
</If>

<If sdk={["vue", "nuxt"]}>
  ```vue {{ del: [1, 4], ins: [2, 5], prettier: false }}
  <Protect :condition="(has) => has({ role: 'admin' }) && isAllowed">
  <Show :when="(has) => has({ role: 'admin' }) && isAllowed">
    <AdminPanel />
  </Protect>
  </Show>
  ```
</If>
