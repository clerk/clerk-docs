import { useAuth, useReverification } from '@clerk/expo'
import { isBiometricCredentialError, useBiometricCredentials } from '@clerk/expo/biometrics'
import { Button } from 'react-native'

export function TransferButton() {
  const { getToken } = useAuth()
  const { reverify } = useBiometricCredentials()

  const transferMoney = useReverification(
    async () => {
      const token = await getToken()
      const res = await fetch('https://example.com/api/reverification-example', {
        method: 'POST',
        headers: { Authorization: `Bearer ${token}` },
      })

      return res.json()
    },
    {
      onNeedsReverification: async ({ level, complete, cancel }) => {
        try {
          const result = await reverify({
            level,
            reason: 'Verify your identity to continue.',
          })

          if (result.status !== 'complete') {
            console.error(
              'Biometric verification is incomplete. Offer another verification method.',
            )
            cancel()
            return
          }

          complete()
        } catch (err) {
          if (
            isBiometricCredentialError(err) &&
            err.code === 'biometric_credential_policy_incompatible'
          ) {
            console.error('Choose another verification method for this credential.')
          }

          cancel()
        }
      },
    },
  )

  const handlePress = async () => {
    try {
      await transferMoney()
    } catch (err) {
      console.error(JSON.stringify(err, null, 2))
    }
  }

  return <Button title="Transfer" onPress={handlePress} />
}
