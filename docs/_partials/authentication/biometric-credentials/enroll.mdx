<If sdk="ios">
  The following example demonstrates how to enroll the current iOS app installation. The `reason` is displayed in the system authentication prompt.

  ```swift {{ filename: 'EnrollBiometricCredentialView.swift', collapsible: true }}
  import SwiftUI
  import ClerkKit

  struct EnrollBiometricCredentialView: View {
    @Environment(Clerk.self) private var clerk

    var body: some View {
      Button("Enable biometric sign-in") {
        Task { await enrollCurrentDevice() }
      }
    }
  }

  extension EnrollBiometricCredentialView {
    func enrollCurrentDevice() async {
      do {
        try await clerk.biometricCredentials.enroll(
          reason: "Use Face ID or Touch ID to sign in."
        )
      } catch {
        // See https://clerk.com/docs/guides/development/custom-flows/error-handling
        // for more info on error handling.
        dump(error)
      }
    }
  }
  ```
</If>

<If sdk="android">
  The following example demonstrates how to enroll the current Android app installation. The `promptTitle` is displayed in the system authentication prompt.

  ```kotlin {{ filename: 'EnrollBiometricCredential.kt', collapsible: true }}
  import com.clerk.api.Clerk
  import com.clerk.api.network.serialization.onFailure
  import com.clerk.api.network.serialization.onSuccess

  suspend fun enrollCurrentDevice() {
    Clerk.biometricCredentials
      .enroll(promptTitle = "Use biometrics to sign in")
      .onSuccess {
        // The current app installation is enrolled.
      }
      .onFailure {
        // See https://clerk.com/docs/guides/development/custom-flows/error-handling
        // for more info on error handling.
      }
  }
  ```
</If>

<If sdk="expo">
  The following example demonstrates how to enroll the current Expo app installation. The `reason` is displayed in the system authentication prompt.

  ```tsx {{ filename: 'components/enroll-biometric-credential.tsx' }}
  import { useBiometricCredentials } from '@clerk/expo'
  import { Button } from 'react-native'

  export function EnrollBiometricCredential() {
    const { enroll } = useBiometricCredentials()

    const enrollCurrentDevice = async () => {
      try {
        await enroll({ reason: 'Use biometrics to sign in.' })
      } catch (err) {
        // See https://clerk.com/docs/guides/development/custom-flows/error-handling
        // for more info on error handling.
        console.error(JSON.stringify(err, null, 2))
      }
    }

    return <Button title="Enable biometric sign-in" onPress={enrollCurrentDevice} />
  }
  ```
</If>
