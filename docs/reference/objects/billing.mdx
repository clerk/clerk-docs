The `Billing` object provides methods for managing billing for a user or Organization.

> [!NOTE]
> If an `orgId` parameter is not provided, the methods will automatically use the current user's ID.

#### Example

<If sdk="js-frontend">
  The `Billing` object is available on the [`Clerk`](/docs/reference/objects/clerk) object.

  ```js {{ filename: 'src/main.js' }}
  import { Clerk } from '@clerk/clerk-js'

  const publishableKey = import.meta.env.VITE_CLERK_PUBLISHABLE_KEY

  // Initialize Clerk with your Clerk Publishable Key
  const clerk = new Clerk(publishableKey)

  // Load Clerk
  await clerk.load()

  // Call the getPaymentAttempt() method
  await clerk.billing.getPaymentAttempt({
    id: 'payment_attempt_123',
  })
  ```
</If>

<If notSdk={["js-frontend", "astro", "vue", "nuxt"]}>
  The `Billing` object is available on the [`Clerk`](/docs/reference/objects/clerk) object. Use the [`useClerk()`](/docs/reference/hooks/use-clerk) hook to access the `clerk.billing` object, and then call the `getPaymentAttempt()` method.

  <If sdk="react">
    ```tsx {{ filename: 'src/pages/Example.tsx' }}
    import { useClerk } from '@clerk/react'
    import { useState } from 'react'

    export default function Example() {
      // Use the useClerk hook to access the clerk object
      const clerk = useClerk()
      const [paymentAttempt, setPaymentAttempt] = useState<Awaited<
        ReturnType<typeof clerk.billing.getPaymentAttempt>
      > | null>(null)

      const handleGetPaymentAttempt = async () => {
        // Call the getPaymentAttempt() method
        const paymentAttempt = await clerk.billing.getPaymentAttempt({
          id: 'payment_attempt_123',

      }

      return (
        <div>
          <button onClick={handleGetPaymentAttempt}>Get payment attempt</button>

        </div>
      )
    }
    ```
